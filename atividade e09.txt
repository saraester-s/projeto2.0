#include<stdio.h> 
int main (void){
  DDRD = DDRD | 0b11100000;  //PINO 7 led verde //saída e entrada (PD7 adicionado: sem isso o LED2 nunca vira saída)
  PORTD = PORTD |   0b00110000; //Habilitar pull up  //Falo se está lógico alto ou baixo, pull up e pull down pra entradas//Soluciono enquanto não mexo nele
  PORTD = PORTD &~(0b11000000);  //Desativar pino 7,6 led verde
 
  for(;;){
 
    int botao = PIND & 0b00100000; //Lê estado pd 5(botao1)
    int botao2 = PIND & 0b00010000; //Lê estado pd4(botão2)
 
    if(botao2 == 0){  //se pressiono botão S2 -> LED2 (PD7)
      PORTD = PORTD | 0b01000000; //PD6 = High -Saída ativa // PORTD |= (1<<PD6)
      _delay_ms(1000);
      PORTD = PORTD &~(0b01000000); //PD6= low       //
      PORTD = PORTD | 0b10000000;
      _delay_ms(1000);
      PORTD = PORTD&~(0b10000000); //(1<<PD7)
    }
 
    if(botao == 0 ){
      PORTD = PORTD | 0b01000000; //PD6 = High -Saída ativa // PORTD |= (1<<PD6)
      _delay_ms(100);
      PORTD = PORTD &~(0b01000000); //PD6= low       //
    }
 
  }
}
